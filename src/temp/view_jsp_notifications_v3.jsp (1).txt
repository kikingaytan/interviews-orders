<%-- 4) JS completo (reemplaza el de la version anterior; misma IIFE propia, con init-once) --%>
<c:if test="${claimNotificationAccessAllowed}">
<script type="text/javascript">
(function () {
    var MAX = 5;                                   // aviso temprano; el servidor manda
    var UPDATE_URL = '${claimNotifUpdateUrl}';
    var BULK_URL = '${claimNotifBulkUrl}';
    var STORE = 'claimNotifPending';

    var serverSubscribed = {
        <c:forEach items="${claimNotificationSubscriptions}" var="e" varStatus="s">
            "<c:out value='${e.key}'/>": "<c:out value='${e.value}'/>"<c:if test="${!s.last}">,</c:if>
        </c:forEach>
    };

    var state = window.__claimNotif;
    if (!state) {
        state = window.__claimNotif = { subscribed: {}, pending: {}, bound: false };
        try { state.pending = JSON.parse(sessionStorage.getItem(STORE) || '{}'); } catch (e) { state.pending = {}; }
    }
    state.subscribed = serverSubscribed;

    function save() { try { sessionStorage.setItem(STORE, JSON.stringify(state.pending)); } catch (e) {} }
    function size(o) { return Object.keys(o).length; }
    function $(id) { return document.getElementById(id); }
    function enc(k, v) { return encodeURIComponent(k) + '=' + encodeURIComponent(v); }

    function say(text, isError) {
        var el = $('claimNotifMsg');
        if (!el) return;
        el.textContent = text || '';
        el.className = 'claim-notif-msg' + (isError ? ' claim-notif-error' : '');
    }

    function limitMsg() { say('You can follow up to ' + MAX + ' claims at a time.', true); }

    // checkboxes visibles y habilitados de una vista ("cards" | "table")
    function boxesIn(scope) {
        var all = document.querySelectorAll('input.claim-notify[data-scope="' + scope + '"]');
        var out = [];
        for (var i = 0; i < all.length; i++) {
            if (!all[i].disabled && all[i].offsetParent !== null) out.push(all[i]);
        }
        return out;
    }

    function addPending(key) {
        if (size(state.subscribed) + size(state.pending) >= MAX) return false;
        state.pending[key] = true;
        return true;
    }

    function refreshBanner() {
        var n = size(state.pending);
        var banner = $('claimNotifBanner');
        if (banner) banner.hidden = (n === 0);
        var label = n + (n === 1 ? ' claim selected' : ' claims selected');
        var count = $('claimNotifCount');
        if (count && count.textContent !== label) count.textContent = label;   // solo si cambia
        if ($('claimNotifSend')) $('claimNotifSend').disabled = (n === 0);
        if ($('claimNotifClear')) $('claimNotifClear').disabled = (n === 0);
    }

    function sync() {
        var boxes = document.querySelectorAll('input.claim-notify');
        for (var i = 0; i < boxes.length; i++) {
            var k = boxes[i].getAttribute('data-key');
            boxes[i].checked = !!(state.subscribed[k] || state.pending[k]);
        }
        var alls = document.querySelectorAll('input.claim-notify-all');
        for (var j = 0; j < alls.length; j++) {
            var list = boxesIn(alls[j].getAttribute('data-scope'));
            var every = list.length > 0;
            for (var m = 0; m < list.length; m++) { if (!list[m].checked) { every = false; break; } }
            alls[j].checked = every;
            alls[j].disabled = (list.length === 0);
        }
        refreshBanner();
    }

    function post(url, params, cb) {
        var xhr = new XMLHttpRequest();
        xhr.open('POST', url, true);
        xhr.setRequestHeader('Content-Type', 'application/x-www-form-urlencoded; charset=UTF-8');
        xhr.onreadystatechange = function () {
            if (xhr.readyState !== 4) return;
            var data = null;
            try { data = JSON.parse(xhr.responseText); } catch (e) {}
            cb(xhr.status === 200 && data ? data : null);
        };
        xhr.send(params);
    }

    function onCheckboxChange(box) {
        var key = box.getAttribute('data-key');
        if (box.checked) {                                    // marcar = pendiente, sin BD
            if (!addPending(key)) { box.checked = false; limitMsg(); return; }
            save(); say('');
            sync();
        } else if (state.pending[key]) {                      // desmarcar un pendiente
            delete state.pending[key]; save(); sync();
        } else if (state.subscribed[key]) {                   // desmarcar uno activo = baja inmediata
            box.disabled = true;
            post(UPDATE_URL, enc('claimKey', key), function (r) {
                box.disabled = false;
                if (r && r.status === 'ok') {
                    delete state.subscribed[key]; say('Email updates turned off for this claim.');
                } else {
                    box.checked = true; say('Could not update. Please try again.', true);
                }
                sync();
            });
        }
    }

    // "Select All on Page": marcar = agrega pendientes hasta el tope; desmarcar = quita solo pendientes
    // (las ya suscritas se dan de baja una por una, no en masa).
    function onSelectAll(all) {
        var list = boxesIn(all.getAttribute('data-scope'));
        if (all.checked) {
            var hitLimit = false;
            for (var i = 0; i < list.length; i++) {
                var key = list[i].getAttribute('data-key');
                if (state.subscribed[key] || state.pending[key]) continue;
                if (!addPending(key)) { hitLimit = true; break; }
            }
            if (hitLimit) limitMsg(); else say('');
        } else {
            for (var j = 0; j < list.length; j++) {
                delete state.pending[list[j].getAttribute('data-key')];
            }
        }
        save(); sync();
    }

    function onSend(btn) {
        var keys = Object.keys(state.pending);
        if (!keys.length) return;
        var modeEl = document.querySelector('input[name="claimNotifMode"]:checked');
        var mode = modeEl ? modeEl.value : 'ALL';
        var body = [enc('mode', mode)];
        for (var i = 0; i < keys.length; i++) body.push(enc('claimKeys', keys[i]));

        btn.disabled = true;
        post(BULK_URL, body.join('&'), function (r) {
            if (!r || r.status !== 'ok') {
                say('Could not save your selection. Please try again.', true);
                refreshBanner();
                return;
            }
            var j, k;
            for (j = 0; j < r.accepted.length; j++) {
                state.subscribed[r.accepted[j]] = mode;
                delete state.pending[r.accepted[j]];
            }
            var limit = false, processed = false;
            for (k in r.rejected) {
                if (!r.rejected.hasOwnProperty(k)) continue;
                delete state.pending[k];
                if (r.rejected[k] === 'LIMIT_REACHED') limit = true;
                if (r.rejected[k] === 'PROCESSED') processed = true;
            }
            save(); sync();
            var parts = [r.accepted.length + ' claim(s) will send email updates.'];
            if (limit) parts.push('Maximum of ' + MAX + ' claims reached; the rest were not added.');
            if (processed) parts.push('Processed claims cannot be tracked.');
            say(parts.join(' '), limit || processed);
        });
    }

    if (!state.bound) {                                   // listeners una sola vez (el script lleva ajaxOnLoad)
        state.bound = true;
        document.addEventListener('change', function (ev) {
            var t = ev.target;
            if (!t || !t.classList) return;
            if (t.classList.contains('claim-notify')) onCheckboxChange(t);
            else if (t.classList.contains('claim-notify-all')) onSelectAll(t);
        });
        document.addEventListener('click', function (ev) {
            if (!ev.target.closest) return;
            var send = ev.target.closest('#claimNotifSend');
            if (send) { onSend(send); return; }
            if (ev.target.closest('#claimNotifClear')) { state.pending = {}; save(); sync(); say(''); }
        });
        var host = document.querySelector('[id$="claimsSummaryContainer"]') || document.body;   // AJUSTAR id real
        new MutationObserver(function () { sync(); }).observe(host, { childList: true, subtree: true });
        // el toggle table/hybrid/card cambia la visibilidad sin tocar el DOM: re-sincronizar los "Select All"
        document.addEventListener('click', function (ev) {
            if (ev.target.closest && ev.target.closest('[data-client-claims-view]')) setTimeout(sync, 0);
        });
    }

    sync();
})();
</script>
</c:if>
